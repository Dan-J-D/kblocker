# syzkaller descriptions for the kblocker module (/sys/kernel/kblocker).
#
# Covers the sysfs control surface, the TLS ClientHello/SNI parser reached
# through TCP traffic (including TCP Fast Open), the UDP/443 path, and the
# file-protection hooks reached through operations on protected paths.
#
# The module must be loaded with hook_all_netns=1: syz-executor runs each
# test process in its own network namespace.

resource fd_kblocker_enabled[fd]
resource fd_kblocker_domains[fd]
resource fd_kblocker_ips[fd]
resource fd_kblocker_pgp[fd]
resource fd_kblocker_key[fd]
resource fd_kblocker_restore[fd]
resource fd_kblocker_state[fd]

# Records of the state file as the kernel writes them, plus damaged and
# legacy (wall-clock "expiry:") forms. A restored session can only be ended
# with the key behind key_hash, which the fuzzer never has, so remaining
# times stay short and legacy expiry times are always in the past.
kblocker_state_line [
	remaining	kblocker_state_remaining
	key_hash	kblocker_state_key_hash
	pgp		kblocker_state_pgp
	domains		kblocker_state_domains
	ips		kblocker_state_ips
	expiry		kblocker_state_expiry
	torn		stringnoz[kblocker_state_prefixes]
	junk		array[int8, 0:16]
] [varlen]

kblocker_state_prefixes = "remaining:", "key_hash:", "domains:", "blocked_ips:", "expiry:"

kblocker_state_pgp {
	tag	stringnoz["pgp_active:1"]
	nl	const[0xa, int8]
} [packed]

kblocker_state_remaining {
	tag	stringnoz["remaining:"]
	val	fmt[dec, int64[0:120]]
	nl	const[0xa, int8]
} [packed]

kblocker_state_key_hash {
	tag	stringnoz["key_hash:"]
	digits	array[flags[kblocker_hex_digits, int8], 62:66]
	nl	const[0xa, int8]
} [packed]

kblocker_state_domains {
	tag	stringnoz["domains:"]
	names	array[kblocker_state_domain, 0:6]
	nl	const[0xa, int8]
} [packed]

kblocker_state_domain {
	name	stringnoz[kblocker_domain_inputs]
	sep	const[0x2c, int8]
} [packed]

kblocker_state_ips {
	tag	stringnoz["blocked_ips:"]
	addrs	array[kblocker_state_ip, 0:6]
	nl	const[0xa, int8]
} [packed]

kblocker_state_ip {
	addr	stringnoz[kblocker_ip_inputs]
	sep	const[0x2c, int8]
} [packed]

kblocker_state_expiry {
	tag	stringnoz["expiry:"]
	val	fmt[dec, int64[0:1700000000]]
	nl	const[0xa, int8]
} [packed]

kblocker_ro_files = "/sys/kernel/kblocker/status", "/sys/kernel/kblocker/remaining", "/sys/kernel/kblocker/block_count", "/sys/kernel/kblocker/key", "/sys/kernel/kblocker/enabled", "/sys/kernel/kblocker/blocked_domains", "/sys/kernel/kblocker/blocked_ips", "/sys/kernel/kblocker/pgp_active"

kblocker_tls_payload [
	hello	kblocker_tls_record
	split	kblocker_tls_record_prefix
	raw	array[int8]
] [varlen]

kblocker_tls_record {
	type	const[0x16, int8]
	ver	int16be[0x300:0x304]
	len	bytesize[hs, int16be]
	hs	kblocker_tls_handshake
} [packed]

# A record header announcing more bytes than are sent, as when a
# ClientHello spans segments.
kblocker_tls_record_prefix {
	type	const[0x16, int8]
	ver	const[0x301, int16be]
	len	int16be[0:0x4000]
	hs	kblocker_tls_handshake
} [packed]

kblocker_tls_handshake {
	type	int8[0:2]
	len_hi	const[0, int8]
	len	bytesize[body, int16be]
	body	kblocker_client_hello
} [packed]

kblocker_client_hello {
	ver	const[0x303, int16be]
	random	array[int8, 32]
	sid_len	len[sid, int8]
	sid	array[int8, 0:32]
	cs_len	bytesize[cs, int16be]
	cs	array[int16be, 1:16]
	cm_len	len[cm, int8]
	cm	array[int8, 1:2]
	ext_len	bytesize[ext, int16be]
	ext	array[kblocker_tls_ext]
} [packed]

kblocker_tls_ext [
	sni	kblocker_ext_sni
	ech	kblocker_ext_ech
	pad	kblocker_ext_padding
	other	kblocker_ext_other
] [varlen]

kblocker_ext_sni {
	type	const[0x0, int16be]
	len	bytesize[list, int16be]
	list	kblocker_sni_list
} [packed]

kblocker_sni_list {
	len	bytesize[names, int16be]
	names	array[kblocker_sni_name, 1:2]
} [packed]

kblocker_sni_name {
	type	int8[0:1]
	len	bytesize[name, int16be]
	name	stringnoz[kblocker_sni_hosts]
} [packed]

# Hosts as they appear on the wire: exact, subdomain, suffix-but-not-
# subdomain, mixed case, trailing dot, and unrelated names.
kblocker_sni_hosts = "youtube.com", "www.youtube.com", "notyoutube.com", "x.com", "netflix.com", "WWW.X.COM.", "reddit.com", "a.b.reddit.com", "example.org", "cloudflare-ech.com", ""

# Encrypted Client Hello (current and draft codepoints)
kblocker_ext_ech {
	type	flags[kblocker_ech_types, int16be]
	len	bytesize[data, int16be]
	data	array[int8, 0:64]
} [packed]

kblocker_ech_types = 0xfe0d, 0xfe0a, 0xfe08

# Moves later extensions past the first segment and past offset 512
kblocker_ext_padding {
	type	const[0x15, int16be]
	len	bytesize[data, int16be]
	data	array[int8, 0:2048]
} [packed]

kblocker_ext_other {
	type	int16be
	len	bytesize[data, int16be]
	data	array[int8, 0:32]
} [packed]
